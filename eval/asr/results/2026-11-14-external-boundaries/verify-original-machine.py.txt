import hashlib
import json
import math
import runpy
import subprocess
import sys
from collections import Counter
from pathlib import Path

base = Path(__file__).resolve().parent
source = base / 'worktrees/asr-languages-1b7e7ea-20261114'
sys.path.insert(0, str(source))
from tools.asr_accuracy import load_corpora, _source
from tools.bench_models import score
from saydraft.models.downloader import installation_ready
from saydraft.models.manifest import get_model
import soundfile as sf

read = lambda path: json.loads(Path(path).read_text(encoding='utf-8'))
sha = lambda path: hashlib.sha256(Path(path).read_bytes()).hexdigest()
root = base / 'external-whisper-boundaries-1b7e7ea-20261114-r1'
process = read(root / 'observer-r1/process.json')
report = read(root / 'native-r1/report.json')
plan = read(root / 'frozen-plan.json')
script = base / 'probe_external_whisper_boundaries_20261114.py'
assert process['native_exit_observed'] and process['returncode'] == 0
assert process['stdout_sha256'] == sha(root / 'observer-r1/stdout.txt')
assert process['stderr_sha256'] == sha(root / 'observer-r1/stderr.txt')
for name in ('report.json', 'cases.jsonl'):
    assert process[name + '_sha256'] == sha(root / 'native-r1' / name)
assert sha(script) == plan['script_sha256'] == report['script_sha256'] == process['script_sha256']
assert sha(root / 'frozen-plan.json') == report['plan_sha256'] == process['plan_sha256']
assert report['source'] == plan['source'] == _source()
assert report['complete'] and len(report['cases']) == 9
assert all(report[key] for key in ('source_unchanged', 'runtime_environment_unchanged', 'pins_unchanged', 'plan_unchanged', 'script_unchanged'))
assert report['model_max_length'] == 448 and report['feature_chunk_length'] == 30
assert installation_ready(get_model('whisper-large-v3-turbo'), base / 'turbo-models/whisper/large-v3-turbo', check_hashes=True)
rows = [json.loads(line) for line in (root / 'native-r1/cases.jsonl').read_text(encoding='utf-8').splitlines()]
assert rows == report['cases'] and sha(root / 'native-r1/cases.jsonl') == report['journal_sha256']
assert [(row['language'], row['id']) for row in rows] == [(row['language'], row['id']) for row in plan['clips']]
clips, receipts = load_corpora([row['path'] for row in report['corpora']])
assert receipts == report['corpora']
lookup = {(row['language'], row['id']): row for row in clips}
policy = runpy.run_path(str(script), run_name='readonly_policy_replay')['boundaries']
totals = {}
for selected, row in zip(plan['clips'], rows):
    clip = lookup[row['language'], row['id']]
    assert sha(selected['baseline_journal']) == selected['baseline_journal_sha256']
    baseline = next(value for value in (json.loads(line) for line in Path(selected['baseline_journal']).read_text(encoding='utf-8').splitlines())
                    if value['id'] == row['id'] and value['model'] == 'whisper-large-v3-turbo')
    assert row['baseline_text'] == baseline['asr_text'] and row['baseline_scores'] == baseline['asr_scores']
    assert row['reference'] == baseline['reference'] == clip['reference']
    assert row['audio_sha256'] == baseline['sha256'] == clip['sha256']
    assert row['scores'] == score(row['reference'], row['text'])
    assert row['baseline_scores'] == score(row['reference'], row['baseline_text'])
    audio, rate = sf.read(clip['audio_path'], dtype='float32')
    assert rate == 16000 and len(audio) == row['audio_samples'] == clip['num_samples']
    expected = policy(audio, plan['policy'])
    assert len(expected) == len(row['parts'])
    for expected_part, part in zip(expected, row['parts']):
        for key in ('start', 'end', 'quiet_boundary'):
            assert part[key] == expected_part[key]
        assert part['boundary_rms'] is None if expected_part['boundary_rms'] is None else math.isclose(part['boundary_rms'], expected_part['boundary_rms'], rel_tol=1e-12)
        for call in part['decode_calls']:
            assert call['max_length'] == 448 and call['prompt_tokens'] == 3
            assert all(0 < count <= 224 for count in call['generated_tokens'])
    group = totals.setdefault(row['language'], {'samples': 0, 'baseline': Counter(), 'candidate': Counter()})
    group['samples'] += 1
    group['baseline'].update(row['baseline_scores'])
    group['candidate'].update(row['scores'])
for group in totals.values():
    assert group['samples'] == 3
    for stage in ('baseline', 'candidate'):
        group[stage] = dict(group[stage])
        group[stage]['wer'] = group[stage]['word_errors'] / group[stage]['reference_words']
        group[stage]['cer'] = group[stage]['character_errors'] / group[stage]['reference_characters']
all_baseline = sum(group['baseline']['word_errors'] for group in totals.values())
all_candidate = sum(group['candidate']['word_errors'] for group in totals.values())
baseline_chars = sum(group['baseline']['character_errors'] for group in totals.values())
candidate_chars = sum(group['candidate']['character_errors'] for group in totals.values())
assert all_candidate > all_baseline and candidate_chars > baseline_chars
result = {'passed': True, 'measurement_integrity': True, 'cases': 9, 'all_source_corpus_audio_baseline_and_score_hashes_verified': True,
          'all_samples_covered_exactly_once': True, 'native_child_exit_observed': True, 'summary': totals,
          'aggregate_development_counts': {'baseline_word_errors': all_baseline, 'candidate_word_errors': all_candidate,
                                           'baseline_character_errors': baseline_chars, 'candidate_character_errors': candidate_chars},
          'decision': 'Reject. Combined word and character error counts worsen; Tamil regressions are substantial. No production segmentation/default change and no holdout inference.',
          'scope': 'Development only; no latency or broad quality acceptance.'}
(root / 'independent-verification.json').write_text(json.dumps(result, indent=2) + '\n', encoding='utf-8')
print(json.dumps(result))

holdout_root = base / 'dense-script-confirmation-corpus-20261114-r1'
state = read(holdout_root / 'state.json')
observer = base / 'dense-script-confirmation-corpus-observer-20261114-r1'
process = read(observer / 'process.json')
assert state['complete'] and not state['errors'] and len(state['languages']) == 3
assert process['native_exit_observed'] and process['returncode'] == 0 and sha(holdout_root / 'state.json') == process['state_sha256']
assert all(sha(observer / (name + '.txt')) == process[name + '_sha256'] for name in ('stdout', 'stderr'))
for item in state['languages']:
    clips, receipts = load_corpora([item['corpus']])
    assert len(clips) == 40 and receipts[0]['sha256'] == item['corpus_sha256']
    assert [row['id'] for row in clips[:20]] == item['excluded_observed_ids']
    assert [row['id'] for row in clips[20:]] == item['heldout_ids']
    assert not set(item['excluded_observed_ids']) & set(item['heldout_ids'])
    assert not item['inference_run']
    assert {str(row['id']): row['sha256'] for row in clips[20:]} == item['heldout_audio_sha256']
holdout_receipt = {'passed': True, 'acquired_clips': 120, 'new_holdout_clips': 60,
                   'baseline_prefix_clips': 60, 'new_holdout_inference_run': False,
                   'source_corpus_tsv_reference_audio_hashes_verified': True, 'child_exit_observed': True,
                   'state_sha256': sha(holdout_root / 'state.json')}
(holdout_root / 'independent-verification.json').write_text(json.dumps(holdout_receipt, indent=2) + '\n', encoding='utf-8')
print(json.dumps(holdout_receipt))
