import hashlib
import importlib.abc
import json
from pathlib import Path
import sys
import time
import numpy as np

class Guard(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, path=None, target=None):
        if fullname.split('.')[0] in ('sounddevice', 'soundcard') or fullname in ('saydraft.app', 'saydraft.compat.migrate'):
            raise AssertionError('Owner audio/profile services are barred')
sys.meta_path.insert(0, Guard())
root=Path('D:/Saydraft-Codex-QA-20261112-01a1cbc7/whisper-mode-20261115-r1')
model=root/'gtcrn_simple.onnx'
assert hashlib.sha256(model.read_bytes()).hexdigest()=='e77603ac0c23dac3227dd2d7135b3a585cbee2679048aecfa886657d3ae1b534'
import sherpa_onnx
if hasattr(sherpa_onnx, 'set_debug'):
    sherpa_onnx.set_debug(False)
from saydraft.privacy.telemetry import disable_sherpa_telemetry
disable_sherpa_telemetry(sherpa_onnx)
config=sherpa_onnx.OfflineSpeechDenoiserConfig(model=sherpa_onnx.OfflineSpeechDenoiserModelConfig(
    gtcrn=sherpa_onnx.OfflineSpeechDenoiserGtcrnModelConfig(model=str(model)),
    num_threads=1,provider='cpu',debug=False))
assert config.validate()
denoiser=sherpa_onnx.OfflineSpeechDenoiser(config)
rows=[]
for n in (512,513,1000,16000,16001,32000):
    audio=(np.sin(np.arange(n,dtype=np.float32)*0.17)*0.01).astype(np.float32)
    started=time.perf_counter()
    output=denoiser(audio,16000)
    result=np.asarray(output.samples,dtype=np.float32)
    rows.append({'input_samples':n,'output_samples':len(result),'sample_rate':output.sample_rate,
        'seconds':time.perf_counter()-started,'finite':bool(np.isfinite(result).all())})
report={'runtime':sherpa_onnx.__version__,'rows':rows,'real_audio_sdk_imported':False}
(root/'native-shape.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
print(json.dumps(report))
