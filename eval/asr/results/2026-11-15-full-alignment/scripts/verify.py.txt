def read(path):return json.loads(Path(path).read_bytes())
def sha(path):
    with Path(path).open('rb') as stream:return hashlib.file_digest(stream,'sha256').hexdigest()
def words(text):
    return ''.join(' ' if unicodedata.category(char)[0] in 'PS' else char
        for char in unicodedata.normalize('NFKC',text).casefold()).split()
def distance(a,b):
    previous=list(range(len(a)+1))
    for j,right in enumerate(b,1):
        current=[j]
        for i,left in enumerate(a,1):current.append(min(current[-1]+1,previous[i]+1,previous[i-1]+(left!=right)))
        previous=current
    return previous[-1]
def score(reference,text):
    expected,actual=words(reference),words(text)
    return {'word_errors':distance(expected,actual),'reference_words':len(expected),
        'character_errors':distance(''.join(expected),''.join(actual)),'reference_characters':len(''.join(expected))}
def wav_samples(path):
    data=Path(path).read_bytes();assert data[:4]==b'RIFF' and data[8:12]==b'WAVE'
    offset,align,length=12,None,None
    while offset+8<=len(data):
        kind,size=struct.unpack_from('<4sI',data,offset);start=offset+8
        assert start+size<=len(data)
        if kind==b'fmt ':
            code,channels,rate,_,align,bits=struct.unpack_from('<HHIIHH',data,start)
            assert channels==1 and rate==16000 and (code,bits,align) in [(1,16,2),(3,32,4)]
        elif kind==b'data':length=size
        offset=start+size+size%2
    assert length and align and length%align==0
    return length//align
