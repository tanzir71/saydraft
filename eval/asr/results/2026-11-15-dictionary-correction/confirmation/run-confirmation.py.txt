"""Observe one bounded native child and retain exact plan/source provenance."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import time

qa=Path(__file__).resolve().parent
repo=qa/'worktrees/dictionary-correction-20261115'
root=qa/'dictionary-correction-20261115-r1'
report=root/'confirmation-r1-process.json'
assert not report.exists()
head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=repo,text=True).strip()
assert not subprocess.check_output(['git','status','--porcelain'],cwd=repo,text=True).strip()
files={}
for name in ('saydraft/asr/biasing.py','tools/dictionary_correction.py','tools/dictionary_confirmation.py'):
    data=(repo/name).read_bytes()
    files[name]={'raw_sha256':hashlib.sha256(data).hexdigest(),'lf_sha256':hashlib.sha256(data.replace(b'\r\n',b'\n')).hexdigest()}
    target=root/'confirmation-source'/name
    target.parent.mkdir(parents=True,exist_ok=True)
    target.write_bytes(data)
plan=root/'confirmation-plan.json'
before=hashlib.sha256(plan.read_bytes()).hexdigest()
args=['C:/Users/tanzir/Desktop/voice_tray/.venv/Scripts/python.exe',str(qa/'run_dictionary_correction_module_20261115.py'),
    str(root/'confirmation-r1-guard.json'),'tools.dictionary_confirmation','--plan',str(plan),'--model-dir',str(qa/'parakeet-models/asr/parakeet-tdt-0-6b-v3-int8'),'--tokenizer-dir',str(qa/'dictionary-biasing-20261114-r1/authentic-tokenizer-r1'),'--dependency-root',str(root/'dependencies'),'--output',str(root/'confirmation-r1')]

env=os.environ.copy()
env.update(HF_HUB_OFFLINE='1',HF_HUB_DISABLE_TELEMETRY='1',DO_NOT_TRACK='1',PYTHONUTF8='1')
assert env.get('GITHUB_ACTIONS')!='true'
started=time.time()
timed_out=False
with (root/'confirmation-r1-stdout.txt').open('xb') as out,(root/'confirmation-r1-stderr.txt').open('xb') as err:
    try:
        completed=subprocess.run(args,cwd=repo,env=env,stdout=out,stderr=err,timeout=600,
            creationflags=subprocess.CREATE_NO_WINDOW)
        code=completed.returncode
    except subprocess.TimeoutExpired:
        code=None
        timed_out=True
stderr=(root/'confirmation-r1-stderr.txt').read_text(encoding='utf-8',errors='replace')
result={'source':head,'source_clean':True,'source_files':files,'plan_sha256':before,
    'plan_unchanged':before==hashlib.sha256(plan.read_bytes()).hexdigest(),
    'child_exit_code':code,'child_exit_observed':not timed_out,'timed_out':timed_out,
    'seconds':time.time()-started,'hotword_encoding_error':'encode hotwords failed' in stderr.lower(),
    'owner_audio_or_input_used':False,'source_unchanged':all(hashlib.sha256((repo/name).read_bytes()).hexdigest()==row['raw_sha256'] for name,row in files.items())}
report.write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
print(json.dumps({key:result[key] for key in ('source','child_exit_code','child_exit_observed','seconds','hotword_encoding_error','source_unchanged','plan_unchanged')}))
raise SystemExit(0 if code==0 and not timed_out and not result['hotword_encoding_error'] else 1)
