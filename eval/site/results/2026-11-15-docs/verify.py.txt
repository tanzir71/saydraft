"""Verify frozen P9.4 source, browser/performance evidence and optional actual portable artifacts."""
import argparse
from collections import Counter
import hashlib
import json
from pathlib import Path, PurePosixPath
import subprocess
import zipfile

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--repo', type=Path, required=True)
parser.add_argument('--site-root', type=Path)
parser.add_argument('--build-root', type=Path)
args = parser.parse_args()
repo = args.repo.resolve()
root = Path(__file__).resolve().parent
read = lambda name: json.loads((root / name).read_bytes())
sha = lambda data: hashlib.sha256(data).hexdigest()
receipt = read('receipt.json')
records = read('files.json')
for name, digest in records.items():
    relative = PurePosixPath(name)
    path = (root / relative).resolve()
    assert not relative.is_absolute() and '..' not in relative.parts and path.is_relative_to(root)
    assert sha(path.read_bytes()) == digest, name
source = receipt['source']
witness = read('source-witness.json')
assert source == witness['source']
queries = ''.join(f'{source}:{name}\n' for name in witness['files']).encode()
objects = subprocess.check_output(['git', 'cat-file', '--batch'], cwd=repo, input=queries)
offset = 0
for name, record in witness['files'].items():
    end = objects.index(b'\n', offset)
    header = objects[offset:end].split()
    assert header[1] == b'blob', name
    size = int(header[2]); committed = objects[end + 1:end + 1 + size]; offset = end + 2 + size
    normalize = lambda data: data if record['binary'] else data.replace(b'\r\n', b'\n')
    assert sha(normalize(committed)) == record['sha256'], name
    assert sha(normalize((repo / name).read_bytes())) == record['sha256'], name
    assert record['modified_before_site_run']
    if record['component_scope']: assert record['modified_before_component_run']
for name, count in [('site-tests-final', 42), ('components-r1', 234)]:
    report = read('raw/' + name + '.json')
    assert report['stats']['expected'] == count
    assert not any(report['stats'][key] for key in ('unexpected', 'skipped', 'flaky'))
    assert len(report['config']['projects']) == 3
assert len([name for name in witness['files'] if name.startswith('site/component-tests/') and name.endswith('.png')]) == 213
assert len(list((root / 'screenshots').glob('*.png'))) == 15
stories = [row for row in read('storybook-index.json')['entries'].values() if row['type'] == 'story']
assert len(stories) == 71
assert len([name for name in witness['files'] if name.startswith('site/src/content/') and name.endswith('.mdx')]) == 22
build_log = (root / 'raw/build-final.txt').read_text(encoding='utf-8')
for phrase in ['43 page(s) built', '0 errors', '0 warnings', '0 hints', '22 content pages']:
    assert phrase in build_log, phrase
assert 'Storybook build completed successfully' in (root / 'raw/storybook-r1.txt').read_text(encoding='utf-8')
contracts = (root / 'raw/contracts-r2.txt').read_text(encoding='utf-8')
assert 'tests 10' in contracts and 'pass 10' in contracts and 'fail 0' in contracts
stale = read('raw/stale-build-receipt.json')
assert len(stale) == 2
for record in stale:
    assert record['build_exit'] != 0 and record['injected_date'] == '2000-01-01' and record['restored_exact_bytes']
    assert sha((repo / record['path']).read_bytes().replace(b'\r\n', b'\n')) == sha(
        subprocess.check_output(['git', 'show', source + ':' + record['path']], cwd=repo).replace(b'\r\n', b'\n'))
for label in ['nested-doc', 'migrated-resource']:
    assert 'valid past/current UTC date within 120 days' in (root / f'raw/stale-{label}-r2.txt').read_text(encoding='utf-8')
urls = Counter()
for row in read('lighthouse-summary.json')['runs']:
    data = (root / 'lighthouse' / row['file']).read_bytes()
    report = json.loads(data)
    assert sha(data) == row['sha256'] and report['finalUrl'] == row['url'] and not report.get('runtimeError')
    assert report['categories']['performance']['score'] >= .95
    assert report['categories']['accessibility']['score'] == report['categories']['seo']['score'] == 1
    assert report['audits']['largest-contentful-paint']['numericValue'] <= 2000
    assert report['audits']['cumulative-layout-shift']['numericValue'] < .05
    urls[report['finalUrl']] += 1
assert len(urls) == 7 and set(urls.values()) == {3} and sum(urls.values()) == 21
assert any(url.endswith('/docs/models/') for url in urls)
assert any(url.endswith('/docs/privacy/verify-offline/') for url in urls)
bundle = read('package/bundle.json'); runtime = read('package/runtime.json')
notice = read('package/notice-receipt.json')
assert bundle['git_commit'] == notice['source'] == source and not bundle['source_dirty']
assert runtime['complete'] and all(runtime['checks'].values())
for flag in ['microphone_used', 'audio_devices_enumerated', 'input_injected', 'owner_clipboard_accessed',
             'active_application_read', 'asr_performed', 'native_firewall_verified']:
    assert runtime[flag] is False, flag
assert not bundle['packaging_gate'] and not bundle['installer_size_gate']
assert notice['source_bundle_zip_exact_match']
assert notice['notice_sha256'] == bundle['file_sha256']['_internal/THIRD_PARTY_NOTICES.md']
assert notice['artifact'] == bundle['artifacts'][notice['zip_name']]
assert not subprocess.check_output(['git', 'diff', '--name-only', receipt['base'], source, '--',
    'saydraft', 'assets', 'requirements.txt', 'Saydraft.spec', 'installer', 'tools/build.ps1',
    'tools/package_audit.py', '.github/workflows'], cwd=repo)
assert receipt['p9_4_accepted'] and receipt['pages'] == 43 and receipt['dated_mdx'] == 22
assert not receipt['full_goal_complete'] and not receipt['owner_accuracy_accepted']
for flag in ['owner_desktop_audio_input_used', 'owner_clipboard_used', 'actions_dispatched', 'pull_requests_created',
             'hosting_deployed', 'installer_qualified', 'runtime_code_changed']:
    assert receipt[flag] is False
assert receipt['reserved_bn_ta_te_recordings_uninferred'] == 60
assert all(code == 0 for code in receipt['observed_terminal_exits'].values())
assert read('outputs.json')['site'] == read('raw/lighthouse-dist-files.json')
if args.site_root:
    for label, directory in [('site', 'dist'), ('storybook', 'storybook-static')]:
        folder = args.site_root / directory
        actual = {path.relative_to(folder).as_posix(): sha(path.read_bytes()) for path in folder.rglob('*') if path.is_file()}
        assert actual == read('outputs.json')[label], label
if args.build_root:
    folder = args.build_root / 'dist/Saydraft'
    actual = {path.relative_to(folder).as_posix(): sha(path.read_bytes()) for path in folder.rglob('*') if path.is_file()}
    assert actual == bundle['file_sha256'] and len(actual) == bundle['files']
    zip_path = args.build_root / 'artifacts' / notice['zip_name']
    assert zip_path.stat().st_size == notice['artifact']['bytes'] and sha(zip_path.read_bytes()) == notice['artifact']['sha256']
    with zipfile.ZipFile(zip_path) as archive:
        names = [name for name in archive.namelist() if not name.endswith('/')]
        # The audit may place the app under one common top-level folder.
        prefix = '' if 'Saydraft.exe' in names else 'Saydraft/'
        assert {name.removeprefix(prefix) for name in names} == set(actual)
        for name in names: assert sha(archive.read(name)) == actual[name.removeprefix(prefix)], name
print(json.dumps({'verified': True, 'source': source, 'records': len(records), 'source_files': len(witness['files']),
    'pages': 43, 'dated_mdx': 22, 'site_checks': 42, 'component_checks': 234, 'node_contracts': 10,
    'lighthouse_runs': 21, 'actual_site_outputs_verified': bool(args.site_root),
    'actual_portable_verified': bool(args.build_root), 'p9_4_accepted': True, 'full_goal_complete': False}))
