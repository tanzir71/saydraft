inputs, events, counts = [], [], []
for track, name in (('microphone', '1003119935936341070.wav'), ('desktop', '10167324587744183095.wav')):
    path = corpus_path.parent / 'audio' / name
    checksum = digest(path)
    clip = next(clip for clip in corpus['clips'] if clip['sha256'] == checksum)
    waveform, rate = sf.read(path, dtype='float32')
    assert rate == 16000 and waveform.ndim == 1 and len(waveform) == clip['num_samples']
    inputs.append({'track': track, 'file': name, 'sha256': checksum, 'frames': len(waveform), 'reference': clip['reference']})
    events.append(CaptureEvent('track', TrackStarted(track, 0.)))
    counts.append(len(waveform))
    # Tracks are file events with compressed delivery, not elapsed live capture.
    events.extend(CaptureEvent('audio', AudioChunk(track, start, waveform[start:start + 1600].copy()))
                  for start in range(0, len(waveform), 1600))
events.insert(1, events.pop(next(index for index, item in enumerate(events) if item.kind == 'track' and item.value.track == 'desktop')))
events.insert(2, CaptureEvent('ready'))
end = CaptureEnd(True, 'stopped', *counts)
events.append(CaptureEvent('end', end))

class PublicCapture:
    def __init__(self):
        self.pending, self.queue = deque(events), deque()
        self.active = self.started = False
    def start(self, *, indicator_ready):
        assert indicator_ready is True and painted and capsule.isVisible()
        self.active = self.started = True
    def take(self, *, limit):
        return [self.queue.popleft() for _ in range(min(limit, len(self.queue)))]
    def poll(self):
        if self.pending:
            for _ in range(min(20, len(self.pending))):
                self.queue.append(self.pending.popleft())
            assert len(self.queue) <= 64
            return None
        self.active = False
        return CaptureOutcome(True, True, 'stopped', 0, end=end)
    def request_stop(self):
        pass
    def stop(self, *, timeout):
        self.active = False
        return CaptureOutcome(True, False, 'forced_stop', 0, True)
