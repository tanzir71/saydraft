"""Verify Whisper mode metadata; optional bundle checks read files without launching apps."""
import argparse
import hashlib
import json
from pathlib import Path, PurePosixPath
import struct
import subprocess
import xml.etree.ElementTree as ET
import zipfile

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--repo', type=Path, required=True)
parser.add_argument('--build-root', type=Path)
args = parser.parse_args()
repo = args.repo.resolve()
root = Path(__file__).resolve().parent
receipt = json.loads((root / 'receipt.json').read_bytes())
manifest = json.loads((root / 'files.json').read_bytes())
sha = lambda data: hashlib.sha256(data).hexdigest()
for name, expected in manifest.items():
    relative = PurePosixPath(name)
    path = (root / relative).resolve()
    assert not relative.is_absolute() and '..' not in relative.parts and path.is_relative_to(root)
    assert sha(path.read_bytes()) == expected, name
source = receipt['source']
witness = json.loads((root / 'raw/source-witness.json').read_bytes())
assert witness['source'] == source and witness['clean'] and witness['source_unchanged_since_final_run_started']
queries = ''.join(f'{source}:{name}\n' for name in witness['files']).encode()
objects = subprocess.check_output(['git', 'cat-file', '--batch'], input=queries, cwd=repo)
offset = 0
for name, record in witness['files'].items():
    assert record['modified_before_final_run_started']
    end = objects.index(b'\n', offset)
    header = objects[offset:end].split()
    assert header[1] == b'blob', name
    size = int(header[2])
    committed = objects[end + 1:end + 1 + size]
    offset = end + 2 + size
    assert sha(committed.replace(b'\r\n', b'\n')) == record['normalized_sha256'], name
    assert sha((repo / name).read_bytes().replace(b'\r\n', b'\n')) == record['normalized_sha256'], name
xml = ET.fromstring((root / 'raw/final-tests-r14.xml').read_bytes())
cases = list(xml.iter('testcase'))
assert not list(xml.iter('failure')) and not list(xml.iter('error'))
assert len(cases) - len(list(xml.iter('skipped'))) == receipt['passed_tests'] == 426
assert len(list(xml.iter('skipped'))) == receipt['skipped_tests'] == 3
assert '426 passed, 3 skipped, 3 deselected' in (root / 'raw/final-tests-r14.txt').read_text()
assert receipt['final_test_exit_observed'] == 0 and receipt['deselected_tests'] == 3
for name in ('final-tests-r14', 'rules', 'adversarial', 'render-r3-1x', 'render-r3-2x'):
    guard = json.loads((root / f'raw/{name}-guard.json').read_bytes())
    assert not any(guard.values()), name
assert '556/556 passed' in (root / 'raw/rules.txt').read_text()
adversarial = json.loads((root / 'raw/adversarial.txt').read_bytes())
assert adversarial['total'] == adversarial['caught'] == 100
assert 'All checks passed!' in (root / 'raw/lint.txt').read_text()
render_count = 0
for scale in (1, 2):
    directory = root / f'renders-{scale}x'
    report = json.loads((directory / 'report.json').read_bytes())
    assert report['platform'] == 'offscreen' and report['scale'] == scale and report['network_guard']
    assert not report['owner_or_download_ports_called'] and not report['owner_accuracy_accepted']
    assert report['renders'] == 32 and len(report['rows']) == 16
    for name, expected in report['source_files'].items():
        assert witness['files'][name]['normalized_sha256'] == expected, name
    for row in report['rows']:
        assert row['no_horizontal_overflow'] and row['unchanged_config_and_history']
        for record in row['files']:
            data = (directory / record['name']).read_bytes()
            assert len(data) == record['bytes'] and sha(data) == record['sha256']
            assert data[:8] == b'\x89PNG\r\n\x1a\n'
            assert struct.unpack('>II', data[16:24]) == (row['width'] * scale, row['height'] * scale)
            render_count += 1
assert render_count == receipt['renders'] == 64
native = json.loads((root / 'raw/native-product-r2.json').read_bytes())
assert native['complete'] and native['source'] == source and not native['source_dirty']
assert native['cache_network_requests'] == 0 and native['python_network_guard']
assert not native['audio_sdk_imported'] and not native['whispered_speech_accuracy_measured']
assert len(native['rows']) == receipt['native_cases'] == 9
assert all(row['input_samples'] == row['output_samples'] for row in native['rows'])
assert any(row['input_samples'] == 400001 for row in native['rows'])
for name in ('runtime', 'enhancement-smoke'):
    report = json.loads((root / f'package/{name}.json').read_bytes())
    assert report['complete'] and report['frozen'] and all(report['checks'].values())
    assert report['audio_devices_enumerated'] is False
assert report['checks']['quiet_speech_native'] and report['enhancement_performed']
audit = json.loads((root / 'package/bundle.json').read_bytes())
assert audit['git_commit'] == source and not audit['source_dirty']
assert not audit['packaging_gate'] and not audit['installer_size_gate']
assert receipt['build_exit_observed'] == receipt['frozen_enhancement_exit_observed'] == 0
assert not receipt['owner_accuracy_accepted'] and not receipt['full_goal_complete']
assert not receipt['p7_4_accepted'] and not receipt['installer_qualified']
bundle_verified = False
if args.build_root:
    build = args.build_root.resolve()
    bundle = build / 'dist/Saydraft'
    for name, expected in audit['file_sha256'].items():
        assert sha((bundle / name).read_bytes()) == expected, name
    assert len([p for p in bundle.rglob('*') if p.is_file()]) == audit['files']
    for name, record in audit['artifacts'].items():
        archive = build / 'artifacts' / name
        assert archive.stat().st_size == record['bytes'] and sha(archive.read_bytes()) == record['sha256']
        with zipfile.ZipFile(archive) as stream:
            assert set(stream.namelist()) == {'Saydraft/' + name for name in audit['file_sha256']}
            for entry, expected in audit['file_sha256'].items():
                assert sha(stream.read('Saydraft/' + entry)) == expected, entry
            for name in ('LICENSE', 'MODEL_CARD.md'):
                expected = (repo / 'assets/licenses/gtcrn' / name).read_bytes()
                assert stream.read('Saydraft/_internal/assets/licenses/gtcrn/' + name) == expected
    assert not any(p.suffix == '.onnx' and 'gtcrn' in p.name for p in bundle.rglob('*'))
    bundle_verified = True
print(json.dumps({'verified': True, 'source': source, 'raw_files': len(manifest), 'tests': 426,
    'renders': render_count, 'native_cases': 9, 'actual_bundle_verified': bundle_verified,
    'owner_accuracy_accepted': False, 'p7_4_accepted': False, 'full_goal_complete': False}))
