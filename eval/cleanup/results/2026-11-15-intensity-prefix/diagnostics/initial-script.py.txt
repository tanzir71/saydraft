def own_meter_dll(name,*args,**kwargs):
    # Reviewed NativeMeter reads only the handles it opens after its own spawn.

    return guarded_dll(name,*args,**kwargs)

calls=[]
class TracedClient(LoopbackClient):
    @contextmanager
    def response_bytes(self,path,**kwargs):
        start=time.perf_counter()
        row={'path':path,'payload_bytes':len(json.dumps(kwargs.get('payload'),ensure_ascii=False).encode()),'chunks':0,'bytes':0}
        try:
            with super().response_bytes(path,**kwargs) as chunks:
                row['headers_seconds']=time.perf_counter()-start
                def traced():
                    for chunk in chunks:
                        if not row['chunks']:row['first_byte_seconds']=time.perf_counter()-start
                        row['chunks']+=1
                        row['bytes']+=len(chunk)
                        yield chunk
                yield traced()
        finally:
            row['total_seconds']=time.perf_counter()-start
            calls.append(row)
