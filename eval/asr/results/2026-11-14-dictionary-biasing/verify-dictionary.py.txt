"""Independently check retained dictionary comparison; no app/SDK/model imports."""
import argparse
from collections import Counter
import hashlib
import json
from pathlib import Path
import re
import subprocess
import unicodedata


def sha(data):
    return hashlib.sha256(data).hexdigest()


def normal(text):
    text=unicodedata.normalize('NFKC',text).casefold()
    return ' '.join(''.join(' ' if unicodedata.category(c).startswith(('P','S')) else c for c in text).split())


def edit(a,b):
    previous=list(range(len(b)+1))
    for i,left in enumerate(a,1):
        current=[i]
        for j,right in enumerate(b,1):
            current.append(min(previous[j]+1,current[-1]+1,previous[j-1]+int(left!=right)))
        previous=current
    return previous[-1]


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--repo',type=Path,required=True)
    parser.add_argument('--evidence',type=Path,default=Path(__file__).resolve().parent)
    parser.add_argument('--corpus-root',type=Path)
    parser.add_argument('--tokenizer-root',type=Path)
    parser.add_argument('--model-root',type=Path)
    args=parser.parse_args()
    root=args.evidence
    def read(name):
        return json.loads((root/name).read_text(encoding='utf-8'))
    files=read('files.json')
    for name,expected in files.items():
        assert '..' not in Path(name).parts and not Path(name).is_absolute()
        assert sha((root/name).read_bytes())==expected,name
    plan=read('plan.json')
    corpus=read('corpus.json')
    assert corpus['dataset']=='google/fleurs' and corpus['license']=='CC-BY-4.0' and corpus['language']=='en'
    assert sha((root/'test.tsv').read_bytes())==corpus['metadata_sha256']
    refs={parts[1]:parts for line in (root/'test.tsv').read_text(encoding='utf-8').splitlines() if len(parts:=line.split('\t'))==7}
    clips={row['id']:row for row in corpus['clips']}
    assert len(clips)==20 and len(corpus['clips'])==20
    for clip in clips.values():
        reference=refs[clip['source_filename']]
        assert int(reference[0])==clip['id'] and reference[2]==clip['reference'] and reference[3]==clip['normalized_reference']
        if args.corpus_root:
            assert sha((args.corpus_root/clip['audio']).read_bytes())==clip['sha256']
    forms={}
    for clip in corpus['clips']:
        for word in re.findall(r'(?<!\w)[A-Za-z]{7,}(?!\w)',clip['reference']):
            forms.setdefault(word.casefold(),word)
    assert plan['terms']==[forms[word] for word in sorted(forms,key=lambda w:(-len(w),w))[:40]]
    assert len(set(plan['terms']))==40 and plan['hotwords_score']==1.5 and plan['max_active_paths']==4
    assert plan['gate']=={'relative_term_recall_gain_min':.3,'no_added_term_false_positives':True,
                         'no_added_word_or_character_errors':True,'all_cases_complete':True}
    process,result=read('process.json'),read('report.json')
    assert process['source']==result['source'] and process['source_clean'] and result['source_clean']
    assert process['child_exit_observed'] and process['child_exit_code']==0 and not process['timed_out']
    assert process['source_unchanged'] and process['plan_unchanged'] and result['plan_unchanged'] and result['model_files_unchanged']
    assert not process['hotword_encoding_error'] and b'encode hotwords failed' not in (root/'stderr.txt').read_bytes().lower()
    assert process['owner_audio_or_input_used'] is False and result['python_network_guard']
    assert result['native_firewall_verified'] is False and result['runtime']=='1.13.4'
    assert process['plan_sha256']==result['plan_sha256']==sha((root/'plan.json').read_bytes())
    assert read('guard.json')=={'real_audio_sdk_imported':False,'owner_profile_module_used':False}
    for name,item in process['source_files'].items():
        raw=(root/'source'/name).read_bytes()
        blob=subprocess.check_output(['git','show',f"{result['source']}:{name}"],cwd=args.repo)
        assert sha(raw)==item['raw_sha256'] and sha(raw.replace(b'\r\n',b'\n'))==item['lf_sha256']
        assert raw.replace(b'\r\n',b'\n')==blob.replace(b'\r\n',b'\n')
    catalog=json.loads(subprocess.check_output(['git','show',f"{result['source']}:saydraft/models/manifest.json"],cwd=args.repo))
    model=next(row for row in catalog['models'] if row['id']==result['model'])
    assert result['model_files']=={row['path']:row['sha256'] for row in model['files']}
    if args.model_root:
        assert all(sha((args.model_root/name).read_bytes())==value for name,value in result['model_files'].items())
    tokenizer=read('tokenizer-receipt.json')
    assert tokenizer==result['tokenizer_receipt'] and tokenizer['all_token_ids_match'] and tokenizer['pieces']==8192
    assert not tokenizer['full_archive_checksum_verified'] and not tokenizer['inference_performed']
    assert tokenizer['tokens_sha256']==result['model_files']['tokens.txt']
    assert sum(row['bytes'] for row in tokenizer['ranges'])==tokenizer['transferred_bytes']==370649
    assert len(tokenizer['ranges'])<=64 and all(row['last']-row['first']+1==row['bytes'] for row in tokenizer['ranges'])
    for member in tokenizer['archive_members']:
        if 'weights' in member['name']:
            assert all(min(row['last']+1,member['offset']+member['size']) <= max(row['first'],member['offset'])
                       for row in tokenizer['ranges'])
    if args.tokenizer_root:
        assert sha((args.tokenizer_root/'tokenizer.model').read_bytes())==tokenizer['tokenizer_sha256']
        assert sha((args.tokenizer_root/'tokenizer.vocab').read_bytes())==tokenizer['vocab_sha256']
    rows=[json.loads(line) for line in (root/'cases.jsonl').read_text(encoding='utf-8').splitlines()]
    assert len(rows)==60 and {row['arm'] for row in rows}==set(plan['arms'])
    counts={}
    termset={term.casefold() for term in plan['terms']}
    for arm in plan['arms']:
        selected=[row for row in rows if row['arm']==arm]
        assert [row['id'] for row in selected]==[row['id'] for row in corpus['clips']]
        total,errors=Counter(),Counter()
        for row in selected:
            clip=clips[row['id']]
            assert row['reference']==clip['reference'] and row['audio_sha256']==clip['sha256'] and row['error'] is None
            ref,hyp=normal(row['reference']),normal(row['text'])
            expected,actual=Counter(ref.split()),Counter(hyp.split())
            terms={'expected':sum(expected[t] for t in termset),
                'matched':sum(min(expected[t],actual[t]) for t in termset),
                'false_positives':sum(max(0,actual[t]-expected[t]) for t in termset)}
            scores={'word_errors':edit(ref.split(),hyp.split()),'reference_words':len(ref.split()),
                'character_errors':edit(ref.replace(' ',''),hyp.replace(' ','')),'reference_characters':len(ref.replace(' ',''))}
            assert terms==row['terms'] and scores==row['scores']
            total.update(terms)
            errors.update(scores)
        assert dict(total)==result['arms'][arm]['terms'] and dict(errors)==result['arms'][arm]['scores']
        counts[arm]=(total['matched'],errors['word_errors'],errors['character_errors'])
    assert counts=={'greedy':(36,55,198),'beam':(37,58,208),'beam_hotwords':(36,63,214)}
    assert result['complete'] and result['relative_term_recall_gain']==0 and not result['development_gate_passed']
    assert not result['ship_or_owner_acceptance'] and not result['release_latency_or_broad_accuracy_accepted']
    print(json.dumps({'verified':True,'source':result['source'],'raw_files':len(files),'terms':40,'cases':60,
        'gain':0,'gate_passed':False,'owner_accuracy_accepted':False,'audio_bytes_verified':bool(args.corpus_root),
        'tokenizer_bytes_verified':bool(args.tokenizer_root),'model_bytes_verified':bool(args.model_root)}))


if __name__=='__main__':
    main()
