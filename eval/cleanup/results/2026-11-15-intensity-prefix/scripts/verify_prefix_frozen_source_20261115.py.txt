source='625c987b134dfabf093651a6054de9c06b33e9fc'

archive=CArchiveReader(str(build/'dist/Saydraft/Saydraft.exe'))

def stable(value):
    if isinstance(value,types.CodeType):
        fields=['co_argcount','co_posonlyargcount','co_kwonlyargcount','co_nlocals','co_stacksize',
                'co_flags','co_code','co_consts','co_names','co_varnames','co_freevars','co_cellvars',
                'co_firstlineno','co_linetable','co_exceptiontable','co_name','co_qualname']
        return {field:stable(getattr(value,field)) for field in fields}
    if isinstance(value,bytes):return {'bytes':value.hex()}
    if isinstance(value,(tuple,list)):return [stable(item) for item in value]
    if isinstance(value,frozenset):return {'frozenset':sorted([stable(item) for item in value],key=lambda item:repr(item))}
    if value is Ellipsis:return {'ellipsis':True}
    return value

rows=[]
paths=['saydraft/ui/'+name+'.py' for name in ['reflow','hub','history_page','styles_page','settings_page',
    'languages_page','model_picker','cleanup_picker','labs_page','command_setup','microphone_page',
    'whisper_mode','privacy_page','dictionary_page','snippets_page']]+['saydraft/ui/theme/controller.py','saydraft/cleanup/llm/client.py','saydraft/cleanup/llm/preparation.py']
for path in paths:
    payload=subprocess.check_output(['git','show',source+':'+path],cwd=repo)
    name=path[:-3].replace('/','.')
    actual=pyz.extract(name)
    expected=compile(payload,path,'exec',dont_inherit=True,optimize=0)
    expected_bytes=json.dumps(stable(expected),sort_keys=True,separators=(',',':')).encode()
    actual_bytes=json.dumps(stable(actual),sort_keys=True,separators=(',',':')).encode()
    assert expected_bytes==actual_bytes,name
    rows.append(dict(module=name,source_path=path,source_sha256=hashlib.sha256(payload).hexdigest(),
        code_fingerprint_sha256=hashlib.sha256(actual_bytes).hexdigest()))
report=dict(source=source,bundle=str(build/'dist/Saydraft/Saydraft.exe'),module_count=len(rows),modules=rows,
    exe_sha256=hashlib.sha256((build/'dist/Saydraft/Saydraft.exe').read_bytes()).hexdigest(),
    method='Compare structural code objects read from actual EXE embedded PYZ against compile(Git source, optimize=0), excluding only co_filename.',
    module_code_executed=False,owner_io_used=False)
(qa/'cleanup-prefix-20261115-r1/frozen-source.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
