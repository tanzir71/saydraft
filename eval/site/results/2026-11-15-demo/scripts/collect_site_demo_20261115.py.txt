source = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=repo, text=True).strip()

    'docs/launch/legacy-site', 'index.html', 'demo', 'vercel.json', '.vercelignore'], cwd=repo).decode().split('\0')
witness = {}
for name in filter(None, paths):
    path = repo / name; data = path.read_bytes()
    binary = path.suffix.lower() in ['.png', '.woff', '.woff2', '.ttf', '.jpg', '.ico', '.gz']
    site_input = name.startswith('site/') or name in ['design/tokens.json', 'tools/generate_tokens.mjs', 'saydraft/__init__.py']
    component_input = name.startswith(('site/src/components/', 'site/src/styles/', 'site/component-tests/', 'site/.storybook/')) or name in ['site/package.json', 'site/package-lock.json', 'site/test-ports.ts', 'site/playwright.components.config.ts']
    if site_input: assert path.stat().st_mtime < starts['site'], name
    if component_input: assert path.stat().st_mtime < starts['components'], name
    witness[name] = {'binary': binary, 'sha256': sha(data if binary else data.replace(b'\r\n', b'\n')),
        'site_input': site_input, 'component_input': component_input}
(target / 'source-witness.json').write_text(json.dumps({'source': source, 'test_starts': starts, 'files': witness}, indent=2) + '\n')

    lcp = report['audits']['largest-contentful-paint']['numericValue']; cls = report['audits']['cumulative-layout-shift']['numericValue']

    else:
        failed = [item['id'] for item in report['categories']['seo']['auditRefs'] if item['weight'] > 0 and report['audits'][item['id']]['score'] != 1]
        assert failed == ['is-crawlable'], failed
    rows.append({'file': path.name, 'url': report['finalUrl'], 'demo_noindex': demo, **scores, 'lcp_ms': lcp, 'cls': cls})
urls = Counter(row['url'] for row in rows)
assert len(rows) == 24 and len(urls) == 8 and set(urls.values()) == {3}
