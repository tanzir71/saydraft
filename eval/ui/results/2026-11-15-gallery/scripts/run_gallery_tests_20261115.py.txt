qt_library = Path(importlib.util.find_spec("PySide6").origin).parent.parent
os.environ['PYTHONPATH'] = os.pathsep.join([str(repo), str(qa / "dictionary-correction-20261115-r1/dependencies"), str(qt_library)])

os.environ.update(QT_STYLE_OVERRIDE="Fusion", PYTEST_DISABLE_PLUGIN_AUTOLOAD="1")

    raise AssertionError("Owner clipboard/desktop/external APIs are barred")
QtWidgets.QApplication.clipboard = forbidden
QtGui.QDesktopServices.openUrl = forbidden
from saydraft.privacy.netguard import NetworkGuard
NetworkGuard.install()
