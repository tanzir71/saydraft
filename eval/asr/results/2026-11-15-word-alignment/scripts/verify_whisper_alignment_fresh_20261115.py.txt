def sha(path):
    value = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            value.update(block)
    return value.hexdigest()
def norm(text):
    return ''.join(' ' if unicodedata.category(c)[0] in 'PS' else c
        for c in unicodedata.normalize('NFKC', text).casefold()).split()
def distance(a, b):
    previous = list(range(len(a) + 1))
    for j, actual in enumerate(b, 1):
        current = [j]
        for i, expected in enumerate(a, 1):
            current.append(min(current[-1] + 1, previous[i] + 1, previous[i - 1] + (actual != expected)))
        previous = current
    return previous[-1]
def score(reference, text):
    expected, actual = norm(reference), norm(text)
    return {'word_errors': distance(expected, actual), 'reference_words': len(expected),
        'character_errors': distance(''.join(expected), ''.join(actual)), 'reference_characters': len(''.join(expected))}
def wave_samples(path):
    data = path.read_bytes()
    assert data[:4] == b'RIFF' and data[8:12] == b'WAVE'
    offset, align, size = 12, None, None
    while offset + 8 <= len(data):
        name, count = struct.unpack_from('<4sI', data, offset)
        start = offset + 8
        assert start + count <= len(data)
        if name == b'fmt ':
            code, channels, rate, _, align, bits = struct.unpack_from('<HHIIHH', data, start)
            assert channels == 1 and rate == 16000 and (code, bits, align) in [(3, 32, 4), (1, 16, 2)]
        elif name == b'data':
            size = count
        offset = start + count + count % 2
    assert align and size and size % align == 0
    return size // align

pairs = {}

for row in report['cases']:
    identity = row['language'], row['id']
    assert identity in selected and row['arm'] in ['control', 'word_alignment'] and not row['error_type']
    pair = pairs.setdefault(identity, {})
    assert row['arm'] not in pair
    pair[row['arm']] = row
    assert row['audio_sha256'] == selected[identity]['audio_sha256']
    assert row['scores'] == score(row['reference'], row['text'])
    totals[row['language']][row['arm']].update(row['scores'])
no_new = True
for identity, pair in pairs.items():
    assert set(pair) == {'control', 'word_alignment'}
    control, candidate = pair['control'], pair['word_alignment']
    assert control['reference'] == candidate['reference'] and control['audio_samples'] == candidate['audio_samples']
    no_new &= not ((not candidate['text'].strip() and bool(control['text'].strip())) or ('\ufffd' in candidate['text'] and '\ufffd' not in control['text']))
replay = {}
for language, values in totals.items():
    control, candidate = values['control'], values['word_alignment']
    replay[language] = {arm: {**values[arm], 'samples': 3} for arm in ['control', 'word_alignment']}
    replay[language]['passes'] = candidate['word_errors'] <= control['word_errors'] and candidate['character_errors'] < control['character_errors']
gate = bool(no_new and all(value['passes'] for value in replay.values()))
assert report['comparison'] == {'languages': replay, 'complete_pairs': True,
    'no_new_empty_or_replacement': no_new, 'development_gate_met': gate}
