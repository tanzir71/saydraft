# Reviewed app.py top-level imports define Qt bridge classes and widgets only.
# Full SaydraftApp/main are never constructed; other owner adapters stay barred.
for finder in sys.meta_path:
    if type(finder).__name__ == 'OwnerIOGuard':
        original_find = finder.find_spec
        finder.find_spec = lambda fullname, path=None, target=None, original=original_find: (
            None if fullname == 'saydraft.app' else original(fullname, path, target))

attempts = []

def forbidden(*args, **kwargs):
    attempts.append('unconfigured fake audio port')
    raise AssertionError('Only explicit fake audio providers are available')

for name in ['InputStream', 'query_devices', 'query_hostapis', 'WasapiSettings']:
    setattr(stub, name, forbidden)
assert 'sounddevice' not in sys.modules
sys.modules['sounddevice'] = stub

def retire_stub():
    intact = sys.modules.get('sounddevice') is stub
    if intact:
        del sys.modules['sounddevice']
    (qa / 'microphone-repair-20261115-r5.stub.json').write_text(json.dumps(dict(
        placeholder_only=True, placeholder_intact=intact, real_sdk_module_executed=False,
        placeholder_retired_before_base_guard=True, unexpected_port_attempts=attempts,
        scope='Reviewed Qt bridge class import; fake audio/foreground/input, offscreen widgets; no normal app startup'
    ), indent=2) + '\n', encoding='utf-8')
