assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=repo, text=True).strip() == source
assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=repo, text=True).strip()

out.mkdir(exist_ok=False)
def copy(src, name):
    target = out / name
    target.parent.mkdir(exist_ok=True, parents=True)
    shutil.copyfile(src, target)
for i in range(1, 12):
    copy(qa / f'meeting-setup-local-r{i:02}.txt', f'checks/r{i:02}.txt')
copy(qa / 'meeting-setup-local-r11-receipt.json', 'checks/r11-receipt.json')
for path in (qa / 'meeting-setup-renders-final-r02').iterdir():
    copy(path, 'renders/' + path.name)
for name in ('bundle.json', 'preflight.json', 'runtime.json', 'requirements-frozen.txt'):
    copy(build / 'audit' / name, 'package/' + name)
copy(build / 'artifacts/SHA256SUMS', 'package/SHA256SUMS')
copy(qa / 'meeting-setup-package-20261114-r2.txt', 'package/build.txt')
copy(qa / 'meeting-setup-package-20261114-r1.txt', 'package/failed-r1/build.txt')
copy(qa / 'meeting-setup-package-20261114-r1/audit/runtime.json', 'package/failed-r1/runtime.json')
copy(qa / 'meeting-setup-icu-diagnostic-r1-runtime.json', 'package/diagnostic-icu-removed-runtime.json')
copy(qa / 'meeting-setup-icu-failure-diagnosis.json', 'package/failed-r1/diagnosis.json')
copy(qa / 'run_meeting_setup_checks_20261114.py', 'drivers/checks.py.txt')
copy(qa / 'render_meeting_setup_20261114.py', 'drivers/render.py.txt')
copy(Path(__file__), 'drivers/collect.py.txt')
copy(qa / 'verify_meeting_setup_20261114.py', 'verify-setup.py.txt')
for label in ('rules', 'adversarial', 'lint'):
    copy(qa / f'meeting-setup-{label}-final.txt', f'checks/{label}.txt')
analysis = (build / 'work/Saydraft/Analysis-00.toc').read_text(encoding='utf-8')
modules = ('saydraft.ui.meeting_setup', 'saydraft.meetings.setup', 'saydraft.meetings.devices', 'saydraft.meetings.asr')
assert all(name in analysis for name in modules)
native = []
def walk(value):
    if isinstance(value, (list, tuple)):
        if len(value) == 3 and isinstance(value[0], str) and value[2] == 'BINARY':
            assert 'windowsapps' not in value[1].lower(), value
            if any(name in value[0].lower() for name in ('icuuc.dll', 'vcruntime140', 'msvcp140')):
                native.append(value)
        for child in value:
            walk(child)
walk(ast.literal_eval(analysis))
(out / 'package/native-origins.json').write_text(json.dumps({'source': source,
    'no_windowsapps_binaries': True, 'selected_origins': native}, indent=2) + '\n', encoding='utf-8')
(out / 'package/setup-modules.json').write_text(json.dumps({'source': source,
    'analysis_sha256': hashlib.sha256(analysis.encode('utf-8')).hexdigest(), 'included_modules': modules}, indent=2) + '\n', encoding='utf-8')
